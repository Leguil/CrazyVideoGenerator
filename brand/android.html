package com.crazylogix.core.designsystem.theme

private val LightColorScheme = lightColorScheme(
    primary = Color(0xFF0E6B3F),
    onPrimary = Color.White,
    primaryContainer = Color(0xFFA3F2BE),
    onPrimaryContainer = Color(0xFF00210E),
    surface = Color(0xFFFBFDF8),
    surfaceContainer = Color(0xFFEFF2EC),
    outline = Color(0xFF717972)
)

private val DarkColorScheme = darkColorScheme(
    primary = Color(0xFF87D5A3),
    onPrimary = Color(0xFF00391D),
    primaryContainer = Color(0xFF00522C),
    onPrimaryContainer = Color(0xFFA3F2BE),
    surface = Color(0xFF111412),
    surfaceContainer = Color(0xFF1D201E),
    outline = Color(0xFF8A938B)
)

@Composable
fun CrazyLogixTheme(
    darkTheme: Boolean = isSystemInDarkTheme(),
    content: @Composable () -> Unit
) {
    val colorScheme = if (darkTheme) DarkColorScheme else LightColorScheme
    MaterialTheme(
        colorScheme = colorScheme,
        typography = CrazyLogixTypography,
        shapes = CrazyLogixShapes,
        content = content
    )
}
